\begin{frame}{Strong Attractors -- Computing $\Reach_i(T, \mathcal{G})$}
\begin{columns}
\column{0.5\textwidth}
    \uncover<+->{
    We build up a working set $A_k$:
    }
    \begin{itemize}[<+->]
        \item Start off with the target set $T$
        \item We can add a state $s$ leading to the working set if:
        \begin{itemize}[<+->]
            \item It belongs to player $i$.
            \item It's a nature state.
            \item It belongs to player $1-i$ and all it's successors are in the working set.
        \end{itemize}
    \end{itemize}
\column{0.5\textwidth}
    \begin{alignat*}{3}
        \uncover<2->{A_0 &\defeq T}\\
        \uncover<4->{A_{k+1} &\defeq A_k&&\cup\{s\in &&\Pred(A_k): s\in S_i\}}\\
        \uncover<5->{        &          &&\cup\{s\in &&\Pred(A_k): s\in S_N\}}\\
        \uncover<6->{        &          &&\cup\{s\in &&\Pred(A_k): s\in S_{1-i}}\\
        \uncover<6->{        &          &&           &&\text{ and } \Succ(s)\subseteq A_k\}}\\
    \end{alignat*} 
    \vspace{-1cm}
        \uncover<7->{\[
        \satr(T, \mathcal{P}) \defeq \bigcup_{k\in\N} A_k\]}
\end{columns}
    
\end{frame}

\begin{frame}{Weak Attractors -- Computing $\Reach_i(T, \mathcal{G})$}
\begin{columns}
\column{0.5\textwidth}
    \only<-4>{
    \uncover<+->{
    We iteratively shrink a working set $B_k$.
    }
    \begin{itemize}[<+->]
        \item Start with the strong attractor of $T$ for player $i$.
        \item Remove states that are in the strong attractor of $S\setminus B_k$ from the graph.
        \item Calculate the strong attractor of $T$ again.
    \end{itemize}
    }
    \only<5->{
        \centering
        \vspace{-3.5mm}

        \includegraphics[width=\textwidth]{figures/watr_illustration.pdf}
    }
    
\column{0.5\textwidth}
\begin{align*}
\uncover<2->{B_0 &\defeq\satr_i(T, \mathcal{P})}\\
\uncover<3->{C_k &\defeq\satr_{1-i}(S\setminus B_k, \mathcal{P}\setminus T)}\\
\uncover<4->{B_{k+1} &\defeq\satr_i(T, \mathcal{P}\setminus C_k)}\\
\end{align*}
\[
\uncover<5->{\watr_i(T,\mathcal{P}) =\bigcap_{k\in\N}B_k}
\]
\end{columns}
    
\end{frame}
